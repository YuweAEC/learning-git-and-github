#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Initialize the 16x2 LCD display at address 0x27. 
// Note: If text does not appear after uploading, change 0x27 to 0x3F.
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Pin Mappings
const int TOUCH_SENSOR_PIN = 2; // Touch Pad Input
const int PUSH_BUTTON_PIN  = 3; // Push Button Input
const int RELAY_PIN        = 4; // Relay Driver Output

void setup() {
  // Define input pins
  pinMode(TOUCH_SENSOR_PIN, INPUT);
  
  // Internal pullup configuration ensures pin stays HIGH until pressed down to GND
  pinMode(PUSH_BUTTON_PIN, INPUT_PULLUP); 
  
  // Set relay control pin as output
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // Start with lock fully secure

  // Fire up the I2C LCD Screen
  lcd.init();
  lcd.backlight();
  
  // System Boot Visualization Banner
  lcd.setCursor(0, 0);
  lcd.print("Smart Lock V1.0");
  lcd.setCursor(0, 1);
  lcd.print("Booting System..");
  delay(2000);
  
  renderSecuredDisplay();
}

void loop() {
  int touchState = digitalRead(TOUCH_SENSOR_PIN);
  int buttonState = digitalRead(PUSH_BUTTON_PIN); 

  // System checks: Sensor goes HIGH when touched OR button drops to LOW when pressed
  if (touchState == HIGH || buttonState == LOW) {
    executeUnlockSequence();
  }
}

void executeUnlockSequence() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print(">> ACCESS OK <<");
  lcd.setCursor(0, 1);
  lcd.print("Lock Retracted ");
  
  digitalWrite(RELAY_PIN, HIGH); // Click the relay shut to inject 12V into lock
  delay(3000);                   // Keep door mechanism open for a 3-second window
  
  digitalWrite(RELAY_PIN, LOW);  // Cut relay power to safely re-lock the solenoid
  renderSecuredDisplay();
}

void renderSecuredDisplay() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("  SYSTEM READY  ");
  lcd.setCursor(0, 1);
  lcd.print("Scan Touch Pad  ");
}
